\begin{frame}{Planned primary analysis: MANCOVA}
\begin{columns}[T,onlytextwidth]
\column{.27\textwidth}
\textbf{Joint outcomes}
\[
\mathbf{Y}_{\mathrm{post}}=
\begin{pmatrix}
FF_{\mathrm{post}}\\ FG_{\mathrm{post}}\\ StA_{\mathrm{post}}\\ GA_{\mathrm{post}}
\end{pmatrix}
\]
\column{.69\textwidth}
\begin{itemize}
\item \textbf{Factor:} condition, with or without the additional Poincar\'e sphere.
\item \textbf{Covariates:} baseline FF, FG, StA, and GA, cognitive load, and spatial reasoning ability to assess RQ 2.
\end{itemize}
\end{columns}
\note{This restores the planned MANCOVA framework. GA is the study's label for structural/Gestalt alignment. \end{frame}

\begin{frame}{Technological Acceptance: Perceived Usefulness and Ease of Use}

\small
\begin{columns}[T,onlytextwidth]

% ---------- Left column ----------
\begin{column}{0.49\textwidth}

\begin{mynote}[Perceived Usefulness]
\begin{itemize}
    \item Interactive graphics made the effects of parameter changes immediately visible.
    \item Hints and direct experimentation supported learning.
    \item Missing examples and definitions reduced the perceived learning value.
\end{itemize}
\end{mynote}

\end{column}

% ---------- Right column ----------
\begin{column}{0.49\textwidth}

\begin{mynote}[Perceived Ease of Use]
\begin{itemize}
    \item Camera resets, scrolling, and perspective controls disrupted interaction.
    \item Sliders were sometimes difficult to adjust precisely.
    \item Participants wanted easier access to instructions, formulas, and previous content.
\end{itemize}
\end{mynote}

\end{column}
\end{columns}

\end{frame}

\begin{frame}{Technological Acceptance: Self-Efficacy and Enjoyment}

\small
\begin{columns}[T,onlytextwidth]

% ---------- Left column ----------
\begin{column}{0.49\textwidth}

\begin{mynote}[Self-Efficacy]
\begin{itemize}
    \item Hints enabled participants to proceed independently.
    \item Some completed the tasks despite uncertainty about matrices and terminology.
    \item Task completion did not necessarily indicate conceptual confidence.
\end{itemize}
\end{mynote}

\end{column}

% ---------- Right column ----------
\begin{column}{0.49\textwidth}

\begin{mynote}[Enjoyment]
\begin{itemize}
    \item Participants enjoyed manipulating parameters and observing graphical changes.
    \item Solving successive tasks created motivation similar to a reward system.
    \item Complex content and awkward controls reduced enjoyment.
\end{itemize}
\end{mynote}
\end{column}
\end{columns}

\begin{frame}{Outlook}
\textbf{Pilot study takeaways:} 
\begin{itemize}
    \item Interactivity and visualization were valued most.
    \item Usability problems and insufficient conceptual guidance were the main barriers.
    \item The comments do not reveal a distinct benefit of the Poincaré sphere.
\end{itemize}
\textbf{Next steps:} 
\begin{itemize}
    \item Collect the remaining expert data and conduct the main study.
    \item Conduct the planned MANCOVA and follow-up tests.
    \item Integrate quantitative outcomes with think-aloud observations to investigate learning mechanisms.
    \item Report findings and implications for model understanding and the  visualizations in physics education.
\end{itemize}
\end{frame}
